(function(){

  const editRecordBtn = $('editRecordBtn'), editDeleteVoiceBtn = $('editDeleteVoiceBtn');

  let selectedDays = new Set();

    try {
      const s = JSON.parse(localStorage.getItem('medSettings_v3') || '{}');
      userName = s.userName || '';

      preferredVoice = s.preferredVoice || 'default';

      settingsSound.checked = soundOn;

    localStorage.setItem('medSettings_v3', JSON.stringify({ userName, soundOn, preferredVoice, theme }));
  }

  function loadReminders() {
    try {
      const raw = localStorage.getItem('medReminders_v3');
      if (raw) {
        const parsed = JSON.parse(raw);
        reminders = parsed.map(r => ({
          id: r.id, time: r.time, message: r.message || 'Time to take your medicine 💊',

          days: r.days || [],

        }));

  }
  function saveReminders() {
    const s = reminders.map(r => ({
      id: r.id, time: r.time, message: r.message,
      voiceEnabled: r.voiceEnabled, repeatMode: r.repeatMode,
      days: r.days, lastDate: r.lastDate
    }));
    localStorage.setItem('medReminders_v3', JSON.stringify(s));
  }

  // ---------- CLOCK & GREETING ----------
  function updateClock() {
    const now = new Date();
    const hh = String(now.getHours()).padStart(2,'0');
    const mm = String(now.getMinutes()).padStart(2,'0');
    const ss = String(now.getSeconds()).padStart(2,'0');
    liveClock.textContent = `${hh}:${mm}:${ss}`;
    const days = ['Sunday','Monday','Tuesday','Wednesday','Thursday','Friday','Saturday'];
    const months = ['January','February','March','April','May','June','July','August','September','October','November','December'];
    liveDate.textContent = `${days[now.getDay()]}, ${months[now.getMonth()]} ${now.getDate()}, ${now.getFullYear()}`;
  }

  function updateGreeting() {
    const h = new Date().getHours();
    let emoji, sub;
    if (h < 12) { emoji = '🌅'; sub = 'Good morning!'; }
    else if (h < 18) { emoji = '☀️'; sub = 'Good afternoon!'; }
    else { emoji = '🌙'; sub = 'Good evening!'; }
    greetingEmoji.textContent = emoji;
    const name = userName ? `, ${userName}` : '';
    greetingText.textContent = `${sub.replace('!','')}${name}!`;
    greetingSub.textContent = 'Hope you are feeling well today 💚';
  }

  // ---------- SPEECH ----------
  function getVoicesAsync() {
    return new Promise(resolve => {
      let v = window.speechSynthesis.getVoices();
      if (v.length) return resolve(v);
      window.speechSynthesis.onvoiceschanged = () => resolve(window.speechSynthesis.getVoices());
      setTimeout(() => resolve(window.speechSynthesis.getVoices()), 700);
    });
  }
  async function getPreferredVoiceObj() {
    const voices = await getVoicesAsync();
    if (!voices.length) return null;
    if (preferredVoice === 'female') {
      return voices.find(v => /female|samantha|zira|karen|tessa|moira|victoria|susan/i.test(v.name))
          || voices.find(v => v.lang.startsWith('en')) || voices[0];
    }
    if (preferredVoice === 'male') {
      return voices.find(v => /male|daniel|alex|david|fred|mark|george/i.test(v.name))
          || voices.find(v => v.lang.startsWith('en')) || voices[0];
    }
    return voices.find(v => v.lang.startsWith('en')) || voices[0];
  }
  async function speakText(text) {
    if (!text || !('speechSynthesis' in window)) return;
    const voice = await getPreferredVoiceObj();
    const u = new SpeechSynthesisUtterance(text);
    if (voice) u.voice = voice;
    u.rate = 0.85; u.pitch = 1.0; u.volume = 1.0;
    window.speechSynthesis.cancel();
    window.speechSynthesis.speak(u);
  }
  function playVoiceOrSpeak(reminder) {
    if (!soundOn) return;
    if (reminder.voiceEnabled && reminder.voiceBlob) {
      const url = URL.createObjectURL(reminder.voiceBlob);
      const a = new Audio(url);
      a.play().catch(() => speakText(reminder.message));
      a.onended = () => URL.revokeObjectURL(url);
    } else if (reminder.voiceEnabled) {
      speakText(reminder.message);
    }
  }

  // ---------- BLUETOOTH ----------
  async function connectBluetooth() {
    if (!navigator.bluetooth) {
      alert('Bluetooth not supported on this device/browser. Reminders will still work locally.');
      return;
    }
    try {
      connectionDot.className = 'dot connecting';
      connectionText.textContent = 'Connecting...';
      connectBtn.disabled = true;
      bluetoothDevice = await navigator.bluetooth.requestDevice({
        acceptAllDevices: true,
        optionalServices: ['battery_service','generic_access','0000ffe0-0000-1000-8000-00805f9b34fb']
      });
      const server = await bluetoothDevice.gatt.connect();
      isConnected = true;
      connectionDot.className = 'dot connected';
      connectionText.textContent = `Connected: ${bluetoothDevice.name || 'Rekabit'}`;
      bluetoothDevice.addEventListener('gattserverdisconnected', () => {
        isConnected = false;
        connectionDot.className = 'dot';
        connectionText.textContent = 'Disconnected';
        connectBtn.disabled = false;
        connectBtn.textContent = '🔗 Connect Rekabit';
      });
      try {
        const s = await server.getPrimaryService('battery_service');
        const c = await s.getCharacteristic('battery_level');
        await c.readValue();
      } catch(e) {}
      connectBtn.textContent = '🔄 Connected';
    } catch(err) {
      console.error(err);
      connectionDot.className = 'dot';
      connectionText.textContent = 'Connection failed';
      connectBtn.disabled = false;
      alert('Could not connect to device. Make sure it is on and nearby.');
    }
  }
  function startKeepAlive() {
    if (connectionInterval) clearInterval(connectionInterval);
    connectionInterval = setInterval(() => {
      if (bluetoothDevice && bluetoothDevice.gatt.connected) {}
      else if (bluetoothDevice) {
        isConnected = false;
        connectionDot.className = 'dot';
        connectionText.textContent = 'Disconnected';
        connectBtn.disabled = false;
        connectBtn.textContent = '🔗 Connect Rekabit';
      }
    }, 5000);
  }

  // ---------- DAY PICKER ----------
  dayPicker.addEventListener('click', e => {
    const chip = e.target.closest('.day-chip');
    if (!chip) return;
    const d = parseInt(chip.dataset.day);
    if (selectedDays.has(d)) { selectedDays.delete(d); chip.classList.remove('active'); }
    else { selectedDays.add(d); chip.classList.add('active'); }

  repeatMode.addEventListener('click', e => {
    const btn = e.target.closest('.mode-btn');

    repeatMode.querySelectorAll('.mode-btn').forEach(b => b.classList.remove('active'));

  // ---------- RECORDING (new) ----------
  async function startRecording() {
    if (isRecording) return;
    if (!navigator.mediaDevices || !window.MediaRecorder) {
      alert('Recording not supported in this browser.');
      return;
    }
    try {
      const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
      mediaRecorder = new MediaRecorder(stream);
      audioChunks = [];
      mediaRecorder.ondataavailable = e => { if (e.data.size) audioChunks.push(e.data); };
      mediaRecorder.onstop = () => {
        voiceBlob = new Blob(audioChunks, { type: 'audio/webm' });
        if (!voiceBlob.size) { voiceBlob = null; voiceStatus.textContent = '⚠️ No audio recorded.'; return; }
        if (audioUrl) URL.revokeObjectURL(audioUrl);
        audioUrl = URL.createObjectURL(voiceBlob);
        voicePreview.src = audioUrl;
        voicePreviewContainer.classList.remove('hidden');
        voiceStatus.textContent = '✅ Voice recorded!';
        stream.getTracks().forEach(t => t.stop());
        isRecording = false;
        recordBtn.classList.remove('recording');
        recordBtn.textContent = '🎤 Record voice';
      };
      mediaRecorder.start();
      isRecording = true;
      recordBtn.classList.add('recording');
      recordBtn.textContent = '⏺️ Recording... (tap to stop)';
      voiceStatus.textContent = 'Recording... speak clearly.';
    } catch(err) {
      alert('Cannot access microphone. Please allow permission.');
    }
  }
  function stopRecording() {
    if (mediaRecorder && mediaRecorder.state === 'recording') mediaRecorder.stop();
    else { isRecording = false; recordBtn.classList.remove('recording'); recordBtn.textContent = '🎤 Record voice'; }
  }
  recordBtn.addEventListener('click', () => { isRecording ? stopRecording() : startRecording(); });
  deleteVoiceBtn.addEventListener('click', () => {
    if (audioUrl) URL.revokeObjectURL(audioUrl);
    audioUrl = null; voiceBlob = null;
    voicePreviewContainer.classList.add('hidden');
    voicePreview.src = '';
    voiceStatus.textContent = 'Voice removed.';
  });

  async function startEditRecording() {
    if (isEditRecording) return;
    try {
      const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
      editMediaRecorder = new MediaRecorder(stream);
      editAudioChunks = [];
      editMediaRecorder.ondataavailable = e => { if (e.data.size) editAudioChunks.push(e.data); };
      editMediaRecorder.onstop = () => {
        editVoiceBlob = new Blob(editAudioChunks, { type: 'audio/webm' });

        if (editAudioUrl) URL.revokeObjectURL(editAudioUrl);
        editAudioUrl = URL.createObjectURL(editVoiceBlob);
        editVoicePreview.src = editAudioUrl;
        editVoicePreviewWrap.classList.remove('hidden');
        editVoiceStatus.textContent = '✅ New voice recorded!';
        stream.getTracks().forEach(t => t.stop());
        isEditRecording = false;
        editRecordBtn.classList.remove('recording');
        editRecordBtn.textContent = '🎤 Record new voice';

    if (editMediaRecorder && editMediaRecorder.state === 'recording') editMediaRecorder.stop();
    else { isEditRecording = false; editRecordBtn.classList.remove('recording'); editRecordBtn.textContent = '🎤 Record new voice'; }

  editRecordBtn.addEventListener('click', () => { isEditRecording ? stopEditRecording() : startEditRecording(); });

    editAudioUrl = null; editVoiceBlob = null;

    editVoicePreview.src = '';

  // ---------- ADD / DELETE / RENDER ----------
  function addReminder() {
    const time = reminderTime.value;
    if (!time) { alert('Please pick a time.'); return; }
    const message = (reminderMessage.value || '').trim() || 'Time to take your medicine 💊';
    if (currentRepeatMode === 'specific' && selectedDays.size === 0) {
      alert('Please select at least one day.');
      return;
    }
    const newR = {
      id: Date.now() + Math.random().toString(36).substr(2,6),
      time, message,
      voiceEnabled: voiceEnabled.checked,
      voiceBlob: (voiceEnabled.checked && voiceBlob) ? voiceBlob : null,
      repeatMode: currentRepeatMode,
      days: currentRepeatMode === 'specific' ? Array.from(selectedDays) : [],
      lastDate: null
    };
    reminders.push(newR);
    saveReminders();

    // reset voice
    if (audioUrl) URL.revokeObjectURL(audioUrl);
    audioUrl = null; voiceBlob = null;
    voicePreviewContainer.classList.add('hidden');
    voicePreview.src = '';
    voiceStatus.textContent = 'Voice saved with the reminder. Record a new one for the next.';
    reminderMessage.value = 'Time to take your medicine 💊';

    renderReminders();
  }

  function deleteReminder(id) {
    if (!confirm('Remove this reminder?')) return;
    reminders = reminders.filter(r => r.id !== id);
    saveReminders();
    renderReminders();
  }

  function dayNames(arr) {
    const names = ['Sun','Mon','Tue','Wed','Thu','Fri','Sat'];
    if (!arr || !arr.length) return '';
    return arr.sort().map(d => names[d]).join(' ');
  }

  function renderReminders() {
    if (!reminders.length) {
      reminderList.innerHTML = `<div class="empty-msg">⏳ No reminders yet. Fill the form above and tap "Add this reminder".</div>`;
      return;
    }
    const sorted = [...reminders].sort((a,b) => a.time.localeCompare(b.time));
    let html = '';
    for (const r of sorted) {
      const hasVoice = r.voiceBlob !== null && r.voiceEnabled;
      const voiceTag = r.voiceEnabled
        ? (hasVoice ? `<span class="voice-tag has-voice">🔊 Custom voice</span>`
                    : `<span class="voice-tag no-voice">🗣️ Text-to-speech</span>`)
        : `<span class="voice-tag no-voice">🔇 Silent</span>`;
      let repeatLabel = '';
      if (r.repeatMode === 'daily') repeatLabel = 'Every day';
      else if (r.repeatMode === 'once') repeatLabel = 'Once';
      else repeatLabel = dayNames(r.days) || 'Specific days';
      html += `
        <div class="reminder-item" data-id="${r.id}">
          <div class="row-top">
            <span class="time">${r.time}</span>
            <span class="days">${repeatLabel}</span>
          </div>
          <div class="message">💬 ${escapeHtml(r.message)}</div>
          <div style="display:flex; gap:8px; flex-wrap:wrap;">${voiceTag}</div>
          <div class="reminder-actions">
            <button class="icon-btn play" data-action="play" data-id="${r.id}">▶️ Play now</button>
            <button class="icon-btn edit" data-action="edit" data-id="${r.id}">✏️ Edit</button>
            <button class="icon-btn delete" data-action="delete" data-id="${r.id}">🗑️ Remove</button>
          </div>
        </div>`;
    }
    reminderList.innerHTML = html;
    reminderList.querySelectorAll('[data-action]').forEach(btn => {
      btn.addEventListener('click', () => {
        const action = btn.dataset.action, id = btn.dataset.id;
        const r = reminders.find(x => x.id === id);
        if (!r) return;
        if (action === 'delete') deleteReminder(id);
        else if (action === 'play') playVoiceOrSpeak(r);
        else if (action === 'edit') openEditModal(id);
      });
    });
  }

  function escapeHtml(s) {
    return String(s).replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
  }

  // ---------- EDIT ----------
  function openEditModal(id) {
    const r = reminders.find(x => x.id === id);
    if (!r) return;
    editingId = id;
    editTime.value = r.time;
    editMessage.value = r.message;
    editVoiceEnabled.checked = r.voiceEnabled;
    editVoiceBlob = r.voiceBlob || null;
    editVoiceStatus.textContent = '';
    if (editAudioUrl) URL.revokeObjectURL(editAudioUrl);
    editAudioUrl = null;
    if (editVoiceBlob) {
      editAudioUrl = URL.createObjectURL(editVoiceBlob);
      editVoicePreview.src = editAudioUrl;
      editVoicePreviewWrap.classList.remove('hidden');
    } else {
      editVoicePreviewWrap.classList.add('hidden');
      editVoicePreview.src = '';
    }
    editModal.classList.remove('hidden');
  }
  function closeEditModal() {
    editModal.classList.add('hidden');
    editingId = null;
    editVoiceBlob = null;
    if (editAudioUrl) { URL.revokeObjectURL(editAudioUrl); editAudioUrl = null; }
    if (isEditRecording) stopEditRecording();
  }
  editCancelBtn.addEventListener('click', closeEditModal);
  editSaveBtn.addEventListener('click', () => {
    if (!editingId) return;
    const r = reminders.find(x => x.id === editingId);
    if (!r) return;
    r.time = editTime.value || r.time;
    r.message = (editMessage.value || '').trim() || 'Time to take your medicine 💊';
    r.voiceEnabled = editVoiceEnabled.checked;
    r.voiceBlob = editVoiceBlob;
    saveReminders();
    renderReminders();
    closeEditModal();
  });

  // ---------- ALARM ----------
  function showAlarm(reminder) {
    currentAlarmId = reminder.id;
    const overlay = document.createElement('div');
    overlay.className = 'alarm-overlay';
    overlay.id = 'alarmOverlay';
    overlay.innerHTML = `
      <div class="alarm-card" id="alarmCard">
        <div class="alarm-face" id="alarmFace">😢</div>
        <div class="alarm-time">${reminder.time}</div>
        <div class="alarm-message">${escapeHtml(reminder.message)}</div>
        <div class="alarm-hint" id="alarmHint">Please take your medicine and tap the button below.</div>
        <div class="alarm-actions">
          <button class="btn btn-success" id="tookBtn" style="flex:1;">✅ I took my medicine</button>
        </div>
      </div>`;
    document.body.appendChild(overlay);

    // play sound
    playVoiceOrSpeak(reminder);

    document.getElementById('tookBtn').addEventListener('click', () => {
      const face = document.getElementById('alarmFace');
      const card = document.getElementById('alarmCard');
      const hint = document.getElementById('alarmHint');
      const actions = overlay.querySelector('.alarm-actions');
      face.textContent = '😊';
      face.classList.add('happy');
      card.classList.add('happy');

      if (e.target === overlay) {

        currentAlarmId = null;
      }
    });
  }

  // ---------- CHECK REMINDERS ----------
  function checkReminders() {
    const now = new Date();
    const hh = String(now.getHours()).padStart(2,'0');
    const mm = String(now.getMinutes()).padStart(2,'0');
    const currentTime = `${hh}:${mm}`;
    const todayStr = now.toDateString();
    const todayDay = now.getDay();

    for (const r of reminders) {
      if (r.time !== currentTime) continue;
      // check repeat
      let shouldFire = false;
      if (r.repeatMode === 'daily') shouldFire = true;
      else if (r.repeatMode === 'once') shouldFire = (r.lastDate === null);
      else if (r.repeatMode === 'specific') shouldFire = (r.days || []).includes(todayDay);

      if (!shouldFire) continue;

      const key = r.id + '-' + todayStr + '-' + r.time;
      if (lastTriggered[key]) continue;
      lastTriggered[key] = true;

      // mark last date
      r.lastDate = todayStr;
      saveReminders();

      // if alarm already open, skip
      if (currentAlarmId) continue;

      showAlarm(r);
      break; // one at a time
    }

    // cleanup
    const keys = Object.keys(lastTriggered);
    for (const k of keys) if (!k.includes(todayStr)) delete lastTriggered[k];
  }

    const msg = (reminderMessage.value || '').trim() || 'This is a test reminder — time to take your medicine 💊';
    const fake = {

      time: (() => { const n = new Date(); return String(n.getHours()).padStart(2,'0') + ':' + String(n.getMinutes()).padStart(2,'0'); })(),
      message: msg,
      voiceEnabled: voiceEnabled.checked,
      voiceBlob: voiceBlob
    };
    showAlarm(fake);
  }
